# v0.7 "Feel Good Update" Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Layer player-facing emotional feedback on top of the existing SwiftUI/SwiftData pixel coloring game — per-tap tactile/visual polish (M1), chapter-clear and completion ceremony (M2), and one-shot milestone/journey/monthly celebrations with a priority-based queue (M3).

**Architecture:** Three feature layers wrapped around a new `Features/Celebration/` module. The core of the module is a pure `CelebrationDetector` (trigger + snapshot + seen set → `[CelebrationEvent]`) plus a stateful `CelebrationCoordinator` (`ObservableObject`) that holds a two-slot queue (fullScreen / toast) and writes `seenKey`s into `PlayerProfile` at display time. Visual polish (cell pop, palette glow, hint pulse) and a 5-tier haptic rate limiter extend existing views directly. A new `AnimationProfile` helper centralizes Reduce Motion / Low Power Mode degradations (3-tier: full / reduced / still-present-but-minimal). A `ParticleEmitterView` (SwiftUI `Canvas` + `TimelineView`, no SpriteKit) renders confetti. `PlayerProfile` gains a `celebrationsSeenData: Data` attribute via SwiftData SchemaV5, with a computed `Set<String>` accessor.

**Tech Stack:** Swift 6 / SwiftUI / SwiftData / AVFoundation, XCTest (unit tests), iOS 17+, Instruments (perf), Xcode 15+.

**Spec reference:** `docs/superpowers/specs/2026-04-21-v0.7-feel-good-update-design.md`

```swift
// PixelColoringGame/Utilities/AnimationProfile.swift
import SwiftUI

struct AnimationProfile: Equatable {
    let reduceMotion: Bool
    let lowPower: Bool

    static let normal = AnimationProfile(reduceMotion: false, lowPower: false)

    var shouldShowParticles: Bool {
        reduceMotion == false
    }

    var cellPopDuration: Double {
        reduceMotion ? 0.1 : 0.25
    }

    var cascadeStep: Double {
        reduceMotion ? 0 : 0.15
    }

    var particleDensityScale: Double {
        if reduceMotion { return 0 }
        return lowPower ? 1.0 / 3.0 : 1.0
    }

    func cascadeDelay(for index: Int) -> Double {
        Double(index) * cascadeStep
    }

    func pop() -> Animation {
        if reduceMotion {
            return .easeOut(duration: 0.1)
        }
        return .spring(response: 0.25, dampingFraction: 0.55)
    }

    func cascadeFade() -> Animation {
        reduceMotion ? .easeOut(duration: 0.2) : .easeOut(duration: 0.3)
    }
}

@MainActor
enum AnimationProfileResolver {
    static func current(
        reduceMotion: Bool,
        lowPower: Bool = ProcessInfo.processInfo.isLowPowerModeEnabled
    ) -> AnimationProfile {
        AnimationProfile(reduceMotion: reduceMotion, lowPower: lowPower)
    }
}
```

```swift
// PixelColoringGame/Utilities/HapticFeedbackRateLimiter.swift
import UIKit

enum HapticTier: CaseIterable {
    case cellFill
    case colorComplete
    case levelComplete
    case chapterClear
    case milestone

    var minimumInterval: TimeInterval {
        switch self {
        case .cellFill: return 0.16
        case .colorComplete: return 0.20
        case .levelComplete: return 0.30
        case .chapterClear: return 0.30
        case .milestone: return 0.30
        }
    }
}

protocol HapticClock {
    func timeIntervalSinceReference() -> TimeInterval
}

struct SystemHapticClock: HapticClock {
    func timeIntervalSinceReference() -> TimeInterval { Date().timeIntervalSinceReferenceDate }
}

@MainActor
protocol HapticGenerating {
    func trigger(_ tier: HapticTier)
}

@MainActor
final class UIKitHapticGenerator: HapticGenerating {
    private let soft = UIImpactFeedbackGenerator(style: .soft)
    private let rigid = UIImpactFeedbackGenerator(style: .rigid)
    private let heavy = UIImpactFeedbackGenerator(style: .heavy)
    private let notification = UINotificationFeedbackGenerator()

    init() {
        soft.prepare(); rigid.prepare(); heavy.prepare(); notification.prepare()
    }

    func trigger(_ tier: HapticTier) {
        switch tier {
        case .cellFill: soft.impactOccurred(intensity: 0.55)
        case .colorComplete: rigid.impactOccurred(intensity: 0.8)
        case .levelComplete: heavy.impactOccurred(intensity: 1.0)
        case .chapterClear: notification.notificationOccurred(.success)
        case .milestone: notification.notificationOccurred(.success)
        }
    }
}

@MainActor
final class HapticFeedbackRateLimiter {
    private var lastFire: [HapticTier: TimeInterval] = [:]
    private let clock: HapticClock
    private let generator: HapticGenerating

    init(
        clock: HapticClock = SystemHapticClock(),
        generator: HapticGenerating = UIKitHapticGenerator()
    ) {
        self.clock = clock
        self.generator = generator
    }

    func fire(_ tier: HapticTier) {
        let now = clock.timeIntervalSinceReference()
        if let last = lastFire[tier], now - last < tier.minimumInterval { return }
        lastFire[tier] = now
        generator.trigger(tier)
    }
}
```

```swift
// PixelColoringGame/Features/Celebration/CelebrationEvent.swift
import Foundation

enum CelebrationSlot: Equatable {
    case fullScreen
    case toast
}

enum CelebrationEvent: Hashable, Identifiable {
    case journeyComplete
    case monthlyComplete(yearMonth: String)
    case chapterClear(chapterID: String)
    case firstPerfect
    case streakMilestone(days: Int)
    case levelsMilestone(count: Int)

    var id: String { seenKey }

    var priority: Int {
        switch self {
        case .journeyComplete: return 100
        case .monthlyComplete: return 90
        case .chapterClear: return 80
        case .firstPerfect: return 70
        case let .streakMilestone(days):
            switch days {
            case 30: return 62
            case 14: return 61
            case 7:  return 60
            default: return 50
            }
        case let .levelsMilestone(count):
            switch count {
            case 100: return 52
            case 50:  return 51
            case 10:  return 50
            default:  return 49
            }
        }
    }

    var seenKey: String {
        switch self {
        case .journeyComplete: return "journeyComplete"
        case let .monthlyComplete(ym): return "monthlyComplete:\(ym)"
        case let .chapterClear(id): return "chapterClear:\(id)"
        case .firstPerfect: return "firstPerfect"
        case let .streakMilestone(days): return "streak:\(days)"
        case let .levelsMilestone(count): return "levels:\(count)"
        }
    }

    var slot: CelebrationSlot {
        switch self {
        case .journeyComplete, .monthlyComplete, .chapterClear:
            return .fullScreen
        case .firstPerfect, .streakMilestone, .levelsMilestone:
            return .toast
        }
    }

    /// Toast auto-dismiss duration in seconds (undefined for fullScreen slot).
    var toastDuration: Double {
        switch self {
        case .firstPerfect: return 4.0
        case .streakMilestone: return 3.5
        case .levelsMilestone: return 3.0
        default: return 0
        }
    }

    /// Returns true if `self` should absorb (suppress) the other event when both are queued.
    func supersedes(_ other: CelebrationEvent) -> Bool {
        if self == other { return false }
        switch self {
        case .journeyComplete:
            return true // journeyComplete absorbs everything else
        case .monthlyComplete:
            switch other {
            case .firstPerfect, .streakMilestone, .levelsMilestone:
                return true
            default:
                return false
            }
        case let .streakMilestone(days):
            if case let .streakMilestone(otherDays) = other {
                return days > otherDays
            }
            return false
        case let .levelsMilestone(count):
            if case let .levelsMilestone(otherCount) = other {
                return count > otherCount
            }
            return false
        case .chapterClear, .firstPerfect:
            return false
        }
    }
}
```

```swift
// PixelColoringGame/Features/Celebration/CelebrationDetector.swift
import Foundation

enum CelebrationTrigger {
    case levelCompleted(storageKey: String, rank: CompletionRank)
    case dailyCompleted(date: Date)
    case appLaunched
}

enum CelebrationDetector {
    static func detect(
        trigger: CelebrationTrigger,
        profile: PlayerProfileSnapshot,
        seen: Set<String>
    ) -> [CelebrationEvent] {
        var results: [CelebrationEvent] = []

        switch trigger {
        case let .levelCompleted(_, rank):
            if profile.isJourneyComplete { results.append(.journeyComplete) }
            for chapterID in profile.completedChapterIDs {
                results.append(.chapterClear(chapterID: chapterID))
            }
            if rank == .perfect, profile.totalPerfectCompletions >= 1 {
                results.append(.firstPerfect)
            }
            appendLevelMilestones(for: profile.totalCompletedLevels, into: &results)
        case .dailyCompleted:
            if profile.dailyCompletionCountForCurrentMonth >= profile.daysInCurrentMonth {
                results.append(.monthlyComplete(yearMonth: profile.currentMonthID))
            }
            appendStreakMilestones(for: profile.currentStreak, into: &results)
            // Daily also allows firstPerfect and level milestones (per spec §4 allow list).
            appendLevelMilestones(for: profile.totalCompletedLevels, into: &results)
        case .appLaunched:
            appendStreakMilestones(for: profile.currentStreak, into: &results)
        }

        return results.filter { !seen.contains($0.seenKey) }
    }

    private static func appendStreakMilestones(for days: Int, into results: inout [CelebrationEvent]) {
        for threshold in [7, 14, 30] where days >= threshold {
            results.append(.streakMilestone(days: threshold))
        }
    }

    private static func appendLevelMilestones(for count: Int, into results: inout [CelebrationEvent]) {
        for threshold in [10, 50, 100] where count >= threshold {
            results.append(.levelsMilestone(count: threshold))
        }
    }
}
```

```swift
                .map { $0.id }
        )
        let isJourneyComplete = catalog.chapters.allSatisfy { chapter in
            chapter.levels.allSatisfy { level in progressValues[level.storageKey]?.completedAt != nil }
        }

        let (monthID, daysInMonth) = Self.monthInfo(calendar: calendar, now: now)
        let completedThisMonth = Self.countCompletedDailyDays(profile: profile, monthID: monthID, calendar: calendar)

        _ = snapshot // reserved for future fields

        return PlayerProfileSnapshot(
            currentStreak: profile.currentStreak,
            bestStreak: profile.bestStreak,
            totalCompletedLevels: totalCompleted,
            totalPerfectCompletions: totalPerfect,
            completedMonthlyRewardMonthIDs: Set(StoredStringSetCodec.decode(profile.completedMonthlyRewardIDsRaw)),
            completedChapterIDs: completedChapters,
            isJourneyComplete: isJourneyComplete,
            currentMonthID: monthID,
            dailyCompletionCountForCurrentMonth: completedThisMonth,
            daysInCurrentMonth: daysInMonth
        )
    }

    private static func monthInfo(calendar: Calendar, now: Date) -> (monthID: String, daysInMonth: Int) {
        let components = calendar.dateComponents([.year, .month], from: now)
        let formatter = DateFormatter()
        formatter.calendar = calendar
        formatter.timeZone = calendar.timeZone
        formatter.dateFormat = "yyyy-MM"
        let monthID = formatter.string(from: now)
        let range = calendar.range(of: .day, in: .month, for: now) ?? (1..<31)
        return (monthID, range.count)
    }

    private static func countCompletedDailyDays(profile: PlayerProfile, monthID: String, calendar: Calendar) -> Int {
        let keys = profile.completedDailyDayKeys
        return keys.filter { $0.hasPrefix(monthID) }.count
    }
```

```swift
// PixelColoringGame/Features/Celebration/CelebrationCoordinator.swift
import Combine
import Foundation

@MainActor
protocol CelebrationSeenStoring: AnyObject {
    func load() -> Set<String>
    func save(_ seen: Set<String>)
}

@MainActor
final class CelebrationCoordinator: ObservableObject {
    @Published private(set) var currentFullScreenEvent: CelebrationEvent?
    @Published private(set) var currentToast: CelebrationEvent?

    private var fullScreenQueue: [CelebrationEvent] = []
    private var toastQueue: [CelebrationEvent] = []
    private let seenStore: CelebrationSeenStoring
    private var seen: Set<String>

    init(seenStore: CelebrationSeenStoring) {
        self.seenStore = seenStore
        self.seen = seenStore.load()
    }

    var isCompletionFlowActive: Bool {
        currentFullScreenEvent != nil || currentToast != nil || !fullScreenQueue.isEmpty || !toastQueue.isEmpty
    }

    // MARK: Enqueue

    func handle(trigger: CelebrationTrigger, profile: PlayerProfileSnapshot) {
        let events = CelebrationDetector.detect(trigger: trigger, profile: profile, seen: seen)
        ingest(events)
    }

    /// Test-only direct ingest path; equivalent to post-detect output.
    func enqueueTestEvents(_ events: [CelebrationEvent]) {
        ingest(events)
    }

    private func ingest(_ events: [CelebrationEvent]) {
        // Sort descending by priority.
        let sorted = events.sorted { $0.priority > $1.priority }
        // Apply supersedes: for each event, drop any other event it supersedes, but record that other's seenKey.
        var surviving: [CelebrationEvent] = []
        for event in sorted {
            var absorbed: [CelebrationEvent] = []
            surviving.removeAll { existing in
                if event.supersedes(existing) {
                    absorbed.append(existing)
                    return true
                }
                return false
            }
            for s in absorbed { markSeen(s.seenKey) }
            if !surviving.contains(where: { $0.supersedes(event) }) {
                surviving.append(event)
            } else {
                // event is absorbed by something already in surviving; mark its seenKey too.
                markSeen(event.seenKey)
            }
        }
        for event in surviving {
            guard !seen.contains(event.seenKey) else { continue }
            switch event.slot {
            case .fullScreen:
                guard !fullScreenQueue.contains(event), currentFullScreenEvent != event else { continue }
                fullScreenQueue.append(event)
            case .toast:
                guard !toastQueue.contains(event), currentToast != event else { continue }
                toastQueue.append(event)
            }
        }
        promoteIfNeeded()
    }

    private func promoteIfNeeded() {
        if currentFullScreenEvent == nil, !fullScreenQueue.isEmpty {
            let next = fullScreenQueue.removeFirst()
            currentFullScreenEvent = next
            markSeen(next.seenKey)
        }
        if currentToast == nil, !toastQueue.isEmpty {
            let next = toastQueue.removeFirst()
            currentToast = next
            markSeen(next.seenKey)
        }
    }

    func dismissCurrentFullScreen() {
        currentFullScreenEvent = nil
        promoteIfNeeded()
    }

    func dismissCurrentToast() {
        currentToast = nil
        promoteIfNeeded()
    }

    private func markSeen(_ key: String) {
        guard !seen.contains(key) else { return }
        seen.insert(key)
        seenStore.save(seen)
    }
}
```

```swift
struct ParticleEmitterView: View {

    @State private var particles: [Particle] = []

    var body: some View {

        } else {
            TimelineView(.animation) { context in
                Canvas { ctx, size in
                    let t = context.date.timeIntervalSince1970
                    for particle in particles {
                        let dt = t - particle.bornAt
                        guard dt <= duration else { continue }
                        let progress = dt / duration
                        let x = particle.x0 + particle.vx * dt
                        let y = particle.y0 + particle.vy * dt + 0.5 * 220 * dt * dt
                        let rect = CGRect(
                            x: x * size.width,
                            y: y * size.height,
                            width: 8,
                            height: 4
                        )
                        var shape = Path(rect)
                        let rotation = particle.rotation + CGFloat(dt) * particle.omega
                        shape = shape.applying(
                            CGAffineTransform(translationX: rect.midX, y: rect.midY)
                                .rotated(by: rotation)
                                .translatedBy(x: -rect.midX, y: -rect.midY)
                        )
                        ctx.fill(shape, with: .color(particle.color.opacity(1 - progress)))
                    }
                }
            }
            .onAppear { seed() }
            .allowsHitTesting(false)
        }

                vy: Double.random(in: -1.4 ... -0.6),
```

```swift
// PixelColoringGame/Features/Celebration/MilestoneToastHost.swift
import SwiftUI
import Combine

final class ToastPolicy {
    private let capSeconds: Double = 8.0
    private var used: Double = 0

    func admit(duration: Double) -> Bool {
        guard used + duration <= capSeconds else { return false }
        used += duration
        return true
    }
    func reset() { used = 0 }
}

struct MilestoneToastHost: View {
    @EnvironmentObject private var coordinator: CelebrationCoordinator
    @Environment(\.hapticFeedback) private var haptics
    @State private var visible: CelebrationEvent?
    @State private var policy = ToastPolicy()
    @State private var dismissWork: DispatchWorkItem?

    var body: some View {
        VStack {
            if let event = visible {
                MilestoneToastView(
                    event: event,
                    onTap: { dismiss(skipPolicyReset: true) },
                    onSwipe: { dismiss(skipPolicyReset: true) }
                )
                .transition(.move(edge: .top).combined(with: .opacity))
            }
            Spacer()
        }
        .animation(.easeOut(duration: 0.3), value: visible)
        .onReceive(coordinator.$currentToast) { newEvent in
            handleIncoming(newEvent)
        }
        .allowsHitTesting(visible != nil)
    }

    private func handleIncoming(_ event: CelebrationEvent?) {
        guard let event else {
            // queue briefly idle → reset policy after a small window
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.5) {
                if coordinator.currentToast == nil { policy.reset() }
            }
            return
        }
        let duration = event.toastDuration
        guard policy.admit(duration: duration) else {
            coordinator.dismissCurrentToast()
            return
        }
        haptics.fire(.milestone)
        withAnimation { visible = event }
        let work = DispatchWorkItem { dismiss(skipPolicyReset: false) }
        dismissWork = work
        DispatchQueue.main.asyncAfter(deadline: .now() + duration + 0.3, execute: work)
    }

    private func dismiss(skipPolicyReset: Bool) {
        dismissWork?.cancel()
        withAnimation { visible = nil }
        coordinator.dismissCurrentToast()
    }
}
```
